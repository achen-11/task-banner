# KB-Task 项目创建实施方案

## 项目概述

基于 task.md 中的需求，我们需要创建一个完整的项目创建功能，支持 kooboo 在线项目和常规任务管理系统两种模式。

## 需求分析

### 核心需求
1. **kb-task 创建项目**：需要新增域名和域名校验
2. **frontend 文件夹**：
   - Vue 空项目 + TailwindCSS 3
   - 配置好的 Vite 和迁移脚本
   - kooboo-cli 创建的 site 目录
3. **项目创建成功**：完整的创建流程和状态管理

### 额外需求（来自评论）
- 创建项目时需要确认是否是 kooboo 在线项目
- 如果不是 kooboo 项目，则作为常规任务管理系统

## 技术方案设计

### 1. 项目类型识别机制

```typescript
// 项目类型枚举
enum ProjectType {
  KOOBOO_ONLINE = 'kooboo_online',
  KOOBOO_SELF_HOSTED = 'kooboo_self_hosted',
  STANDARD_WEB = 'standard_web'
}

// 项目类型检测服务
class ProjectTypeDetectionService {
  async detectProjectType(input: {
    domain?: string;
    description?: string;
    requirements?: string[];
  }): Promise<ProjectType> {
    // 如果明确指定了域名，默认认为是 kooboo 项目
    if (input.domain) {
      return ProjectType.KOOBOO_ONLINE;
    }

    // 根据描述和需求关键词判断
    const text = [input.description, ...(input.requirements || [])].join(' ').toLowerCase();

    const koobooKeywords = [
      'kooboo', 'cms', '内容管理', '站点', '网站',
      'kb-', '内容发布', '网页设计'
    ];

    const hasKoobooKeywords = koobooKeywords.some(keyword => text.includes(keyword));

    return hasKoobooKeywords ? ProjectType.KOOBOO_ONLINE : ProjectType.STANDARD_WEB;
  }
}
```

### 2. 域名验证机制

```typescript
// 域名验证服务
class DomainValidationService {
  private readonly domainPatterns = {
    // 标准域名格式：example.com
    standard: /^[a-zA-Z0-9]([a-zA-Z0-9-]{0,61}[a-zA-Z0-9])?(\.[a-zA-Z]{2,})+$/,

    // 支持中文字符的域名
    chinese: /^[\u4e00-\u9fa5a-zA-Z0-9]([-\u4e00-\u9fa5a-zA-Z0-9]{0,61}[\u4e00-\u9fa5a-zA-Z0-9])?(\.[a-zA-Z]{2,})+$/,

    // IPv4 地址
    ipv4: /^(\d{1,3}\.){3}\d{1,3}$/,

    // localhost
    localhost: /^localhost(:\d+)?$/
  };

  async validateDomain(domain: string): Promise<{
    isValid: boolean;
    type: 'standard' | 'chinese' | 'ipv4' | 'localhost' | 'invalid';
    suggestions?: string[];
  }> {
    const trimmedDomain = domain.trim().toLowerCase();

    // 检查各种域名格式
    for (const [type, pattern] of Object.entries(this.domainPatterns)) {
      if (pattern.test(trimmedDomain)) {
        // 进一步验证域名是否可用
        const isAvailable = await this.checkDomainAvailability(trimmedDomain);
        return {
          isValid: true,
          type: type as any,
          isAvailable
        };
      }
    }

    // 如果无效，提供建议
    return {
      isValid: false,
      type: 'invalid',
      suggestions: this.generateDomainSuggestions(trimmedDomain)
    };
  }

  private async checkDomainAvailability(domain: string): Promise<boolean> {
    try {
      // 使用 DNS 查询或第三方 API 检查域名可用性
      // 这里可以集成域名查询服务
      const dns = require('dns').promises;
      try {
        await dns.resolve4(domain);
        return false; // 域名已被占用
      } catch {
        return true; // 域名可用
      }
    } catch (error) {
      console.warn('域名检查失败:', error);
      return true; // 默认认为可用
    }
  }

  private generateDomainSuggestions(input: string): string[] {
    const suggestions: string[] = [];
    const cleaned = input.replace(/[^a-zA-Z0-9\u4e00-\u9fa5]/g, '');

    // 基于输入生成建议
    if (cleaned.length > 0) {
      suggestions.push(`${cleaned}.com`);
      suggestions.push(`${cleaned}.cn`);
      suggestions.push(`${cleaned}.net`);
      suggestions.push(`my${cleaned}.com`);
    }

    return suggestions;
  }
}
```

### 3. 项目创建服务

```typescript
// 项目创建服务
class ProjectCreationService {
  constructor(
    private projectRepository: ProjectRepository,
    private domainService: DomainValidationService,
    private typeDetectionService: ProjectTypeDetectionService,
    private koobooService: KoobooService,
    private workspaceService: WorkspaceService
  ) {}

  async createProject(projectData: CreateProjectRequest): Promise<Project> {
    // 1. 检测项目类型
    const projectType = await this.typeDetectionService.detectProjectType({
      domain: projectData.domain,
      description: projectData.description,
      requirements: projectData.requirements
    });

    // 2. 验证域名（如果是 kooboo 项目）
    let domainValidation: DomainValidationResult | null = null;
    if (projectType !== ProjectType.STANDARD_WEB && projectData.domain) {
      domainValidation = await this.domainService.validateDomain(projectData.domain);

      if (!domainValidation.isValid) {
        throw new Error(`域名验证失败: ${domainValidation.suggestions?.join(', ') || '无效域名'}`);
      }
    }

    // 3. 创建项目记录
    const project = await this.projectRepository.create({
      name: projectData.name,
      description: projectData.description,
      type: projectType,
      domain: projectData.domain,
      domainValidation,
      status: 'creating',
      createdBy: projectData.createdBy
    });

    try {
      // 4. 根据项目类型执行不同的创建流程
      if (projectType === ProjectType.KOOBOO_ONLINE) {
        await this.createKoobooProject(project);
      } else {
        await this.createStandardProject(project);
      }

      // 5. 更新项目状态
      await this.projectRepository.update(project.id, {
        status: 'active',
        createdAt: new Date(),
        updatedAt: new Date()
      });

      return project;

    } catch (error) {
      // 创建失败，清理资源
      await this.cleanupFailedProject(project.id);
      await this.projectRepository.update(project.id, {
        status: 'failed',
        error: error.message,
        updatedAt: new Date()
      });

      throw error;
    }
  }

  private async createKoobooProject(project: Project): Promise<void> {
    // 1. 创建工作区
    const workspacePath = await this.workspaceService.createWorkspace(project.id);

    // 2. 创建 kooboo-cli 项目
    await this.koobooService.initializeKoobooProject(workspacePath, {
      projectName: project.name,
      domain: project.domain!
    });

    // 3. 创建前端项目
    await this.createFrontendProject(workspacePath);

    // 4. 创建站点
    const siteResult = await this.koobooService.createSite(workspacePath, {
      domain: project.domain!,
      name: project.name
    });

    // 5. 更新项目信息
    await this.projectRepository.update(project.id, {
      workspacePath,
      koobooSiteId: siteResult.siteId,
      koobooSiteUrl: siteResult.url,
      status: 'active'
    });
  }

  private async createStandardProject(project: Project): Promise<void> {
    // 1. 创建工作区
    const workspacePath = await this.workspaceService.createWorkspace(project.id);

    // 2. 创建标准前端项目
    await this.createFrontendProject(workspacePath, {
      isKooboo: false,
      projectName: project.name
    });

    // 3. 更新项目信息
    await this.projectRepository.update(project.id, {
      workspacePath,
      status: 'active'
    });
  }

  private async createFrontendProject(
    workspacePath: string,
    options: { isKooboo?: boolean; projectName?: string }
  ): Promise<void> {
    const frontendPath = path.join(workspacePath, 'frontend');

    // 使用 Vite 创建 Vue 项目
    await this.executeCommand(`npm create vue@latest ${frontendPath}`, {
      cwd: workspacePath
    });

    // 安装依赖
    await this.executeCommand('npm install', {
      cwd: frontendPath
    });

    // 安装 TailwindCSS
    await this.executeCommand('npm install -D tailwindcss@latest postcss@latest autoprefixer@latest', {
      cwd: frontendPath
    });

    // 配置 TailwindCSS
    await this.configureTailwindCSS(frontendPath);

    // 如果是 kooboo 项目，配置构建脚本
    if (options.isKooboo) {
      await this.configureKoobooBuild(frontendPath);
    }

    // 创建基础配置文件
    await this.createBaseConfigFiles(frontendPath, options);
  }

  private async configureTailwindCSS(frontendPath: string): Promise<void> {
    const tailwindConfig = `
/** @type {import('tailwindcss').Config} */
export default {
  content: [
    "./index.html",
    "./src/**/*.{vue,js,ts,jsx,tsx}",
  ],
  theme: {
    extend: {},
  },
  plugins: [],
}
    `;

    await fs.writeFile(
      path.join(frontendPath, 'tailwind.config.js'),
      tailwindConfig
    );

    const postcssConfig = `
export default {
  plugins: {
    tailwindcss: {},
    autoprefixer: {},
  },
}
    `;

    await fs.writeFile(
      path.join(frontendPath, 'postcss.config.js'),
      postcssConfig
    );

    // 更新 CSS 文件
    const cssContent = `
@tailwind base;
@tailwind components;
@tailwind utilities;
    `;

    const mainCssPath = path.join(frontendPath, 'src/style.css');
    if (await fs.pathExists(mainCssPath)) {
      const existingCss = await fs.readFile(mainCssPath, 'utf-8');
      await fs.writeFile(mainCssPath, cssContent + existingCss);
    } else {
      await fs.writeFile(mainCssPath, cssContent);
    }
  }

  private async configureKoobooBuild(frontendPath: string): Promise<void> {
    const viteConfig = `
import { defineConfig } from 'vite'
import vue from '@vitejs/plugin-vue'
import { resolve } from 'path'

export default defineConfig({
  plugins: [vue()],
  resolve: {
    alias: {
      '@': resolve(__dirname, './src')
    }
  },
  build: {
    outDir: '../site/static/js',
    rollupOptions: {
      input: {
        main: resolve(__dirname, 'src/main.js')
      },
      output: {
        entryFileNames: 'app.js',
        chunkFileNames: '[name].[hash].js'
      }
    }
  },
  server: {
    port: 5173,
    proxy: {
      '/api': {
        target: 'http://localhost:3000',
        changeOrigin: true
      }
    }
  }
})
    `;

    await fs.writeFile(
      path.join(frontendPath, 'vite.config.ts'),
      viteConfig
    );

    // 创建构建脚本
    const packageJsonPath = path.join(frontendPath, 'package.json');
    const packageJson = JSON.parse(await fs.readFile(packageJsonPath, 'utf-8'));

    packageJson.scripts = {
      ...packageJson.scripts,
      'build:kooboo': 'vite build && npm run copy-to-kooboo',
      'copy-to-kooboo': 'node scripts/copy-to-kooboo.js',
      'dev:kooboo': 'vite build --watch & npm run copy-to-kooboo --watch'
    };

    await fs.writeFile(packageJsonPath, JSON.stringify(packageJson, null, 2));

    // 创建复制脚本
    await this.createCopyScript(frontendPath);
  }

  private async createCopyScript(frontendPath: string): Promise<void> {
    const scriptsDir = path.join(frontendPath, 'scripts');
    await fs.mkdir(scriptsDir, { recursive: true });

    const copyScript = `
const fs = require('fs');
const path = require('path');

const sourceDir = path.join(__dirname, '../dist');
const targetJsDir = path.join(__dirname, '../site/static/js');
const targetCssDir = path.join(__dirname, '../site/static/css');

// 创建目标目录
if (!fs.existsSync(targetJsDir)) {
  fs.mkdirSync(targetJsDir, { recursive: true });
}
if (!fs.existsSync(targetCssDir)) {
  fs.mkdirSync(targetCssDir, { recursive: true });
}

// 复制 JS 文件
const copyJsFiles = () => {
  if (fs.existsSync(sourceDir)) {
    const files = fs.readdirSync(sourceDir);
    files.forEach(file => {
      const srcPath = path.join(sourceDir, file);
      const destPath = path.join(targetJsDir, file);
      if (fs.statSync(srcPath).isFile()) {
        fs.copyFileSync(srcPath, destPath);
      }
    });
  }
};

// 复制 CSS 文件（如果存在）
const copyCssFiles = () => {
  const cssSourceDir = path.join(__dirname, 'dist');
  if (fs.existsSync(cssSourceDir)) {
    const files = fs.readdirSync(cssSourceDir);
    files.forEach(file => {
      if (file.endsWith('.css')) {
        const srcPath = path.join(cssSourceDir, file);
        const destPath = path.join(targetCssDir, file);
        fs.copyFileSync(srcPath, destPath);
      }
    });
  }
};

// 主函数
if (require.main === module) {
  copyJsFiles();
  copyCssFiles();
  console.log('文件已复制到 kooboo site 目录');
}

// 导出供其他脚本使用
module.exports = { copyJsFiles, copyCssFiles };
    `;

    await fs.writeFile(
      path.join(scriptsDir, 'copy-to-kooboo.js'),
      copyScript
    );
  }

  private async createBaseConfigFiles(
    workspacePath: string,
    options: { isKooboo?: boolean; projectName?: string }
  ): Promise<void> {
    const frontendPath = path.join(workspacePath, 'frontend');

    // 创建 .claude 目录和配置
    const claudeDir = path.join(frontendPath, '.claude');
    await fs.mkdir(claudeDir, { recursive: true });

    const claudeConfig = {
      projectName: options.projectName || 'kb-task-project',
      isKooboo: options.isKooboo || false,
      workspacePath,
      buildCommand: options.isKooboo ? 'npm run build:kooboo' : 'npm run build',
      devCommand: options.isKooboo ? 'npm run dev:kooboo' : 'npm run dev'
    };

    await fs.writeFile(
      path.join(claudeDir, 'project-context.json'),
      JSON.stringify(claudeConfig, null, 2)
    );

    // 创建环境配置文件
    await this.createEnvFiles(frontendPath);

    // 创建基础文档
    await this.createDocumentation(frontendPath);
  }

  private async createEnvFiles(frontendPath: string): Promise<void> {
    const envContent = `
# 环境变量
VITE_APP_TITLE=${process.env.VITE_APP_TITLE || 'KB-Task Project'}
VITE_API_URL=${process.env.VITE_API_URL || 'http://localhost:3000'}
VITE_WS_URL=${process.env.VITE_WS_URL || 'ws://localhost:3000'}
    `;

    await fs.writeFile(path.join(frontendPath, '.env'), envContent);

    // 创建示例环境文件
    await fs.writeFile(
      path.join(frontendPath, '.env.example'),
      envContent
    );
  }

  private async createDocumentation(frontendPath: string): Promise<void> {
    const readmeContent = `
# ${path.basename(frontendPath)}

## 项目说明

这是一个基于 Vite + Vue 3 的前端项目。

### 开发环境

\`\`\`bash
npm install

\`\`\`

### 构建部署

\`\`\`bash

\`\`\`

### 项目结构

- \`public/\` - 静态资源
- \`dist/\` - 构建输出

  }

  private async cleanupFailedProject(projectId: string): Promise<void> {
    try {
      const project = await this.projectRepository.findById(projectId);
      if (project?.workspacePath) {
        await this.workspaceService.removeWorkspace(projectId);
      }
    } catch (error) {
      console.error('清理失败项目失败:', error);
    }
  }
}
```

### 4. 工作区管理服务

```typescript
// 工作区管理服务
class WorkspaceService {
  private readonly baseWorkspacePath = '/tmp/kb-workspaces';

  async createWorkspace(projectId: string): Promise<string> {
    const workspacePath = path.join(this.baseWorkspacePath, projectId);

    // 创建工作区目录
    await fs.mkdir(workspacePath, { recursive: true });

    // 设置权限
    await fs.chmod(workspacePath, 0o755);

    // 创建子目录结构
    await fs.mkdir(path.join(workspacePath, 'logs'), { recursive: true });
    await fs.mkdir(path.join(workspacePath, 'backups'), { recursive: true });

    // 创建项目配置文件
    await this.createWorkspaceConfig(workspacePath, projectId);

    return workspacePath;
  }

  async removeWorkspace(projectId: string): Promise<void> {
    const workspacePath = path.join(this.baseWorkspacePath, projectId);

    if (await fs.pathExists(workspacePath)) {
      await fs.rm(workspacePath, { recursive: true, force: true });
    }
  }

  private async createWorkspaceConfig(workspacePath: string, projectId: string): Promise<void> {
    const config = {
      projectId,
      createdAt: new Date().toISOString(),
      nodeVersion: process.version,
      platform: process.platform
    };

    await fs.writeFile(
      path.join(workspacePath, '.workspace.json'),
      JSON.stringify(config, null, 2)
    );
  }
}
```

```vue
<script setup lang="ts">
import { ref, computed, reactive, onMounted } from 'vue'
import { ElMessage } from 'element-plus'
import type { FormInstance, FormRules } from 'element-plus'

// 接口定义
interface ProjectForm {
  projectType: 'auto' | 'kooboo' | 'standard'
  name: string
  description: string
  domain: string
  domainPrefix: string
  requirements: string
  template: string
  options: string[]
}

interface DomainValidation {
  isValid: boolean
  type: string
  isAvailable?: boolean
  suggestions?: string[]
}

interface ProgressLog {
  timestamp: Date
  message: string
  type: 'info' | 'success' | 'warning' | 'error'
}

const props = defineProps<{
  visible: boolean
}>()

const emit = defineEmits<{
  'update:visible': [value: boolean]
  'project-created': [project: any]
}>()

// 表单引用
const formRef = ref<FormInstance>()

// 表单数据
const form = reactive<ProjectForm>({
  projectType: 'auto',
  name: '',
  description: '',
  domain: '',
  domainPrefix: 'https://',
  requirements: '',
  template: 'default',
  options: ['创建示例页面', '配置 ESLint']
})

// 状态
const submitting = ref(false)
const checkingDomain = ref(false)
const domainError = ref(false)
const domainValidation = ref<DomainValidation | null>(null)
const showProgress = ref(false)
const currentStep = ref(0)
const progressPercentage = ref(0)
const progressStatus = ref<'success' | 'exception' | 'warning' | ''>('success')
const progressMessage = ref('')
const progressLogs = ref<ProgressLog[]>([])

// 计算属性
const isFormValid = computed(() => {
  return form.name.trim() &&
         form.description.trim() &&
         form.requirements.trim() &&
         (!shouldShowDomainField || (form.domain.trim() && domainValidation.value?.isValid))
})

const shouldShowDomainField = computed(() => {
  return form.projectType === 'kooboo' || form.projectType === 'auto'
})

const isKoobooProject = computed(() => {
  return form.projectType === 'kooboo' ||
         (form.projectType === 'auto' && form.domain.trim())
})

// 表单验证规则
const rules: FormRules<ProjectForm> = {
  name: [
    { required: true, message: '请输入项目名称', trigger: 'blur' },
    { min: 2, max: 100, message: '项目名称长度应在 2-100 个字符之间', trigger: 'blur' }
  ],
  description: [
    { required: true, message: '请输入项目描述', trigger: 'blur' },
    { min: 10, max: 500, message: '项目描述长度应在 10-500 个字符之间', trigger: 'blur' }
  ],
  requirements: [
    { required: true, message: '请输入项目需求', trigger: 'blur' },
    { min: 10, max: 1000, message: '项目需求长度应在 10-1000 个字符之间', trigger: 'blur' }
  ],
  domain: [
    {
      validator: (rule, value, callback) => {
        if (!shouldShowDomainField.value) {
          callback()
          return
        }
        if (!value.trim()) {
          callback(new Error('请输入项目域名'))
          return
        }
        callback()
      },
      trigger: 'blur'
    }
  ]
}

// 项目模板
const availableTemplates = ref([
  {
    id: 'default',
    name: '默认模板',
    description: '基础的 Vue 3 + Vite 项目结构'
  },
  {
    id: 'dashboard',
    name: '仪表板模板',
    description: '包含数据可视化和管理界面'
  },
  {
    id: 'cms',
    name: 'CMS 模板',
    description: '内容管理系统基础结构'
  },
  {
    id: 'blog',
    name: '博客模板',
    description: '个人博客系统结构'
  }
])

// 方法
const handleSubmit = async () => {
  if (!formRef.value) return

  try {
    await formRef.value.validate()
    submitting.value = true
    showProgress.value = true
    currentStep.value = 0
    progressPercentage.value = 0
    progressLogs.value = []

    await executeProjectCreation()

    ElMessage.success('项目创建成功！')
    emit('project-created', { /* 项目信息 */ })
    handleClose()

  } catch (error) {
    console.error('项目创建失败:', error)
    ElMessage.error(error.message || '项目创建失败')

    progressLogs.value.push({
      timestamp: new Date(),
      message: error.message,
      type: 'error'
    })
  } finally {
    submitting.value = false
  }
}

const executeProjectCreation = async () => {
  try {
    // 步骤1: 验证信息
    await addProgressLog('验证项目信息...', 'info')
    currentStep.value = 1
    progressPercentage.value = 10

    // 步骤2: 创建工作区
    await addProgressLog('创建项目工作区...', 'info')
    // 调用项目创建服务
    currentStep.value = 2
    progressPercentage.value = 30

    // 步骤3: 初始化项目
    await addProgressLog('初始化项目结构...', 'info')
    currentStep.value = 3
    progressPercentage.value = 50

    // 步骤4: 创建前端项目
    await addProgressLog('创建前端项目...', 'info')
    currentStep.value = 4
    progressPercentage.value = 70

    // 步骤5: 配置 Kooboo (如果是 kooboo 项目)
    if (isKoobooProject.value) {
      await addProgressLog('配置 Kooboo 集成...', 'info')
      currentStep.value = 5
      progressPercentage.value = 85
    }

    // 完成
    await addProgressLog('项目创建完成！', 'success')
    currentStep.value = 6
    progressPercentage.value = 100
    progressStatus.value = 'success'

  } catch (error) {
    progressStatus.value = 'exception'
    await addProgressLog(`创建失败: ${error.message}`, 'error')
    throw error
  }
}

const validateDomain = async () => {
  if (!form.domain.trim()) {
    domainValidation.value = null
    return
  }

  checkingDomain.value = true
  domainError.value = false

  try {
    // 调用域名验证服务
    const result = await domainValidationService.validateDomain(form.domain)
    domainValidation.value = result

    if (!result.isValid) {
      domainError.value = true
    }
  } catch (error) {
    domainValidation.value = {
      isValid: false,
      type: 'invalid',
      suggestions: []
    }
  } finally {
    checkingDomain.value = false
  }
}

const checkDomainAvailability = async () => {
  if (!form.domain.trim()) return

  checkingDomain.value = true

  try {
    const result = await domainValidationService.validateDomain(form.domain)
    domainValidation.value = result

    if (result.isValid) {
      const message = result.isAvailable
        ? '域名可用 ✅'
        : '域名已被占用 ⚠️'

      ElMessage.info(message)
    } else {
      ElMessage.error('域名格式无效')
    }
  } catch (error) {
    ElMessage.error('域名检查失败')
  } finally {
    checkingDomain.value = false
  }
}

const addProgressLog = (message: string, type: ProgressLog['type']) => {
  progressLogs.value.push({
    timestamp: new Date(),
    message,
    type
  })
}

const formatTime = (date: Date) => {
  return date.toLocaleTimeString()
}

const handleClose = () => {
  emit('update:visible', false)
  resetForm()
}

const resetForm = () => {
  form.projectType = 'auto'
  form.name = ''
  form.description = ''
  form.domain = ''
  form.domainPrefix = 'https://'
  form.requirements = ''
  form.template = 'default'
  form.options = ['创建示例页面', '配置 ESLint']

  domainValidation.value = null
  domainError.value = false
  progressLogs.value = []
}

// 监听项目类型变化
watch(() => form.projectType, (newType) => {
  if (newType !== 'kooboo' && newType !== 'auto') {
    form.domain = ''
    domainValidation.value = null
    domainError.value = false
  }
})
```
